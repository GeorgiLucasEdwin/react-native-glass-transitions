App.tsx
import { Appearance, Switch, View, useColorScheme } from 'react-native';
import { GlassTransitionProvider, useGlassSource } from 'react-native-glass-transitions';

export default function App() {
  return (
    <GlassTransitionProvider preset="glass">
      <Settings />
    </GlassTransitionProvider>
  );
}

function Settings() {
  const dark = useColorScheme() === 'dark';
  const source = useGlassSource();

  return (
    <View {...source.props}>
      <Switch
        value={dark}
        onValueChange={(on) =>
          source.transition(() => Appearance.setColorScheme(on ? 'dark' : 'light'))
        }
      />
    </View>
  );
}
